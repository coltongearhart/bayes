Using independent samples

Now we can simulate from the posterior.

# solve for posterior parameters
b_hat <- solve(t(X) %*% X) %*% t(X) %*% y
post_alpha <- (nrow(X) - ncol(X)) / 2
post_gamma <- t(y - X %*% b_hat) %*% (y- X %*% b_hat) / 2

# initialize items
m <- 10000
post_beta <- matrix(data = 0, nrow = m , ncol = 4)
colnames(post_beta) <- paste0("b",0:3)

# generate samples
# -> initialize needed items
post_sigma2 <- 1 / rgamma(n = m, shape = post_alpha, rate = post_gamma)
X_t_Xinv <- solve(t(X) %*% X)
for(i in 1:m) {
  
  post_beta[i,] <- MASS::mvrnorm(1, b_hat, post_sigma2[i] * X_t_Xinv)

}

Using Stan

Method 1: Using individual predictors

data {
  int<lower=0> n; // sample size
  vector[n] y;
  vector[n] x1;
  vector[n] x2;
  vector[n] x3;
  int<lower=0> n2; // sample size of the future observations
  vector[n2] x1_new;
  vector[n2] x2_new;
  vector[n2] x3_new;
}
parameters {
  real beta0;
  real beta1;
  real beta2;
  real beta3;
  real<lower=0> sigma2;
}
model {
  y ~ normal(beta0 + beta1 * x1 + beta2 * x2 + beta3 * x3, sqrt(sigma2)); // data distribution
  target += -log(sigma2); // prior
}
generated quantities {
  vector[n2] y_pred;
  for (i in 1:n2) {
    y_pred[i] = normal_rng(beta0 + beta1 * x1_new[i] + beta2 * x2_new[i] + beta3 * x3_new[i], sqrt(sigma2)); //the y values predicted by the model
  }
}

Bayesian linear regression models: Analysis with conjugate priors

data {
  int n; // the number of observations
  int n2; // the size of the new_X matrix
  int p; // the number of columns in the model matrix
  real y[n]; // the response
  matrix[n,p] X; // the model matrix
  matrix[n2,p] X_new; // the matrix for the predicted values
}
parameters {
  vector[p] beta; // the regression parameters
  real<lower=0> sigma2; // variance
}
model {
  y ~ normal(X * beta, sqrt(sigma2));

Practice

Setup:

data-haircut.csv is a data set about the gender and cost of last haircut they paid for. Could you fit a model to decide if female student spend more on haircut on average? In this data set, some NAs are existing for the response (money spent on haircut). You need to remove these; additionally, the predictor variable gender needs to be recoded as 0 and 1 to fit the model.

Solution:

With gender as an indicator variable, the model statements become:

\[ \begin{align*} y = \beta_0 + \beta_1 I_{Male} + \epsilon\\ \end{align*} \]

\[ \begin{align*} \Rightarrow y &= \beta_0 + \epsilon \quad \text{(Female)}\\ y &= \beta_0 + \beta_1 + \epsilon \quad \text{(Female)}\\ \end{align*} \]
